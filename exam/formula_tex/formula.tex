\subsection*{MLR}
\begin{flushleft}
\textbf{Linear Model: } $Y = X\beta + \epsilon$, with $\mathbb{E}[\epsilon] = 0$ and $\text{Var}(\epsilon) = \sigma^2 I_{n \times n}$

\vspace{1ex}
\textbf{Slope: } $\hat{\beta} = (X^TX)^{-1}X^Ty$

\vspace{1ex}
\textbf{Simple LR: } $\hat{\beta}_1 = \frac{\sum_{i=1}^{n}(x_i - \bar{x})y_i}{\sum_{i=1}^{n}(x_i - \bar{x})^2}$, \quad $\hat{\beta}_0 = \bar{y} - \hat{\beta}_1 \bar{x}$

\section{Model Selection}
\begin{minipage}[t]{0.45\textwidth}
    \subsection*{Subset Selection}
    Given model \(M_0\) for null model, \(M_p\) for full model, and \(M_k\) for \(k\) predictors,
    \begin{itemize}
        \item \textbf{Best Subset Selection:} For \(k = 1, 2, \dots, p\), fit all \(\binom{p}{k}\) models and choose the best one. 
        \item \textbf{Forward Stepwise Selection:} Start with \(M_0\) and add one variable at a time.
        \item \textbf{Backwards Stepwise Selection:} Start with \(M_p\) and remove one variable at a time.  
    \end{itemize}
\end{minipage}
\hspace{0.05\textwidth}

\section{Polynomial Regression and Spline}
\begin{minipage}[t]{0.45\textwidth}
    \subsection*{Polynomial Regression}
        \[
            y_i = \beta_0 + \beta_1 x_i + \beta_2 x_i^2 + \ldots + \beta_d x_i^d + \epsilon_i
        \]
        \text{Giving K knots, we have K+1 intervals.}
        \[
            C_0(X) + C_1(X) + \ldots + C_{K-1}(X) + C_K(X) = 1.
        \]
        \[
            y_i = \beta_0 + \beta_1C_1(x_i) + \beta_2C_2(x_i) + \ldots + \beta_KC_K(x_i) + \epsilon_i
        \]
\end{minipage}
\hspace{0.05\textwidth} 
\begin{minipage}[t]{0.45\textwidth}
    \subsection*{Cubic Spline}
        A cubic spline with K knots,
        \[
            y_i = \beta_0 + \beta_1C_1(x_i) + \beta_2C_2(x_i) + \ldots + \beta_KC_K(x_i) + \epsilon_i.
        \]
        Truncated polynomial basis functions:
        \[
            h(x, \xi ) =(x - \xi)^3_+ = \begin{cases}
                (x - \xi)^3, & x > \xi \\
                0, & x \leq \xi
            \end{cases}
        \]
\end{minipage}
\vspace{1em}
\begin{itemize}
    \item Freedom of Cubic Spline: \(K + 4\) degrees of freedom.
    \item  A degree-d spline is a piecewise degree-d polynomial with continuity
    in derivatives up to degree d − 1.
    \item To fit a cubic spline with K knots, we fit a linear regression with an
    intercept and the K + 3 predictors:
    \(
        x, x^2, x^3, h_1(x, \xi_1), h_2(x, \xi_2), \ldots, h_K(x, \xi_K).
    \)
    \item  A natural cubic spline with K knots uses K degrees of freedom
\end{itemize}

\subsection*{Smoothing Splines}
We can try and ensure that g(x) is smooth by choosing g(x) that minimizes
\(
    \sum_{i=1}^{n}(y_i - g(x_i))^2 + \lambda \int g''(x)^2 dx
\).\\
\textbf{Effective Degrees of Freedom: }
\(
    df_{\lambda} = \sum_{i = 1}^{n}[S_\lambda]_{ii}, \ \text{the trace of } S_\lambda.
\)

\section{MARS and CART}
\begin{minipage}[t]{0.45\textwidth}

\[
    \hat{f}(x) = \sum_{i=1}^{k}c_iB_i(x)
\]

\begin{itemize}

    \item a hinge function \(max(0, x - c) \text{ or } max(0,  c -x)\)

\end{itemize}

\section{PCA}
Consider $p$ predictors \(X_1, X_2, \dots, X_p\),

\begin{minipage}[t]{0.45\textwidth}
    \subsection*{First PC}
    \begin{align*}
        &Z_1 = \varPhi_{11} X_1 + \varPhi_{21} X_2 + \dots + \varPhi_{p1} X_p, \quad \sum_{j=1}^{p}{\varPhi_{p1}}^2 = 1 \\
        &Z_1 = {\alpha_1}^TX \\
        &var({\alpha_1}^TX) = {\alpha_1}^T\sum \alpha_1
    \end{align*}
    \subsection*{Lagarange}
    \begin{align*}
        &{\alpha_1}^T\sum \alpha_1 - \lambda({\alpha_1}^T\sum \alpha_1 - 1) \\
        &\text{Differentiating w.r.t. } \alpha_1 \text { and setting to zero}, \Rightarrow \sum \alpha_1 = \lambda \alpha_1 \\
        & \Rightarrow \Max {\alpha_1}^T\sum \alpha_1 = \Max {\alpha_1}^T \lambda \alpha_1 = \Max \lambda {\alpha_1}^T  \alpha_1 = \Max \lambda
    \end{align*}
\end{minipage}
\hfill
\begin{minipage}[t]{0.45\textwidth}
    \subsection*{Second PC}
    \begin{align*}
        &\Max {\alpha_2}^T \sum \alpha_2, \quad \text{uncorrelated with } {\alpha_1}^T X.\\
        & cov({\alpha_1}^T X, {\alpha_2}^T X) = {\alpha_1}^T \sum \alpha_2
                                              = {\alpha_2}^T \sum \alpha_1
                                              = {\alpha_2}^T \lambda \alpha_1
                                              = \lambda {\alpha_2}^T \alpha_1
    \end{align*}
    \subsection*{Lagarange}
    \begin{align*}
        &{\alpha_2}^T \sum \alpha_2 - \lambda ({\alpha_2}^T \alpha_2 - 1) - \varPhi({\alpha_2}^T \alpha_2 - 0) \\
        &\text{Differentiating w.r.t. } \alpha_2 \text{ and setting to zero}, \\
        & \Rightarrow \sum \alpha_2 - \lambda \alpha_2 - \varphi \alpha_1 = 0. \\
        & \Rightarrow {\alpha_1}^T \sum \alpha_2 - \lambda {\alpha_1}^T \alpha^2 - \varPhi {\alpha_1}^T \alpha_1 = 0, \\
        & \Rightarrow \varPhi = 0 \\
        & \Rightarrow \sum \alpha_2 = \lambda \alpha_2.
    \end{align*}
\end{minipage}

\section{K-means Clustering}
\subsection*{Notation}
Let \(C_1, \dots, C_k\) denote sets containing indices of each observation in each cluster.
\begin{itemize}
    \item \(C_1 \cup C_2 \cup \dots \cup C_k = \{1,2,\dots,n\}\)
    \item \(C_k \cap {C_K}' = \varnothing \) for all \(k \neq k'\).
\end{itemize}
\subsection*{Clustering}
\begin{minipage}[t]{0.45\textwidth}
    \[
    \min_{C_1, C_2, \dots, C_k} 
    \left\{
    \sum_{k=1}^{K} \frac{1}{|C_k|} 
    \sum_{i, i' \in C_k} 
    \sum_{j=1}^{p} (x_{ij} - x_{i'j})^2
    \right\}
    \]
    Note that
    \[
    \frac{1}{|C_k|}\sum_{i, i' \in C_k} \sum_{j=1}^{p} (x_{ij} - x_{i'j})^2
    = 2 \sum_{i \in C_k}\sum_{j=1}^{p} (x_{ij} - \bar{x}_{kj})^2,
    \]
    where 
    \[
    \bar{x}_{kj} = \frac{1}{|C_k|}\sum_{i \in C_k}x_{ij}.
    \]
\end{minipage}
\hfill
\begin{minipage}[t]{0.45\textwidth}
    \textbf{Algorithm}
    \begin{enumerate}
        \item Randomly assign a number of 1 to \(K\) to each observation.
        \item Iterate until no changes in cluster assignment:
        \begin{enumerate}
            \item For each cluster, calculate the centroid. The centroid is the vector of the sample mean of the p features for observations in the cluster.
            \item Assign each observation to the cluster of the centroid that is closest. 
        \end{enumerate}
    \end{enumerate}
\end{minipage}

\section{Multi-dimensional Scaling}
\begin{minipage}[t]{0.45\textwidth}
    \subsection*{Scaling Procedures}
\begin{itemize}
    \item Classical scaling: dissimilarities are Euclidean distances.
    \item Ordinal scaling: uses only the order of the dissimilarity measure.
\end{itemize}
\end{minipage}
\hfill
\begin{minipage}[t]{0.45\textwidth}
    \subsection*{Dissimilarity Coefficients}
    The dissimilarity coefficient between two points s and r must satisfy the following constraints:
    \begin{enumerate}
        \item \(d_{sr} \geq 0\) for every \(r\) and \(s\).
        \item \(d_{rr} = 0\) for every \(r\). 
        \item \(d_{sr} = d_{rs}\) for every \(r\) and \(s\).
        \item  \(d_{rt} + d_{ts} \geq d_{rs}\) for every \(r, s\) and \(t\) (Only if A metric satisfies 1-3).
    \end{enumerate}
\end{minipage}
\subsection*{Dissimilarity Coefficients}
\begin{minipage}[t]{0.32\textwidth}
    \textbf{Euclidean}
    \[
    d_{rs} = \left(\sum_{i=1}^{p}(x_{ri} - x_{si})^2\right)^\frac{1}{2}
    \]
\end{minipage}
\hfill
\begin{minipage}[t]{0.32\textwidth}
    \textbf{Manhattan}
    \[
    d_{rs} = \sum_{i=1}^{p}|x_{ri} - x_{si}|
    \]
\end{minipage}
\hfill
\begin{minipage}[t]{0.32\textwidth}
    \textbf{Minkowski}
    \[
    d_{rs} = \left(\sum_{i=1}^{p}|x_{ri} - x_{si}|^R\right)^\frac{1}{R}
    \]
\end{minipage}

    \]
\end{minipage}
\subsection*{Classical Scaling}
\begin{multicols}{3}
    Consider 
\[
B = XX^T
\]
The \((r, s)\)th term of B
\[
b_{rs} = \sum_{j=1}^{p}x_{rj}x{sj}.
\]
\textbf{Euclidean Distances}
\[
d_{rs}^2 = \sum_{j=1}^{p} (x_{rj} - x_{sj})^2 = b_{rr} + b_{ss} - 2b_{rs}.
\]
Summing Over \textit{r}, \textit{s}, and Both
\[
\sum_{r=1}^{n} d_{rs}^2 = T + n b_{ss}
\]
\[
\sum_{s=1}^{n} d_{rs}^2 = n b_{rr} + T
\]
\[
\sum_{r=1}^{n} \sum_{s=1}^{n} d_{rs}^2 = 2nT
\]

\noindent where
\[
T = \sum_{r=1}^{n} b_{rr},
\]
i.e., the trace of \( B \). \\
Rearranging gives
\[
b_{rs} = -\frac{1}{2} \left[ d_{rs}^2 - d_{r\bullet}^2 - d_{s\bullet}^2 + d_{\bullet\bullet}^2 \right].
\]
\end{multicols}

\section{The EM Algorithm}
\begin{multicols}{2}
\subsection*{Example Using Two Normals}
Consider a model with two normally distributed groups. For ease of explanation assume that each group has the same variance.
\begin{align*}
    &Y_1 \sim N(\mu_1, \sigma^2),\\
    &Y_1 \sim N(\mu_1, \sigma^2),\text{ and}\\
    &Y = (1 - \bigtriangleup)Y_1 + \bigtriangleup Y_2,
\end{align*}
where \(\bigtriangleup in \{0,1\}\) and
\[
P(\bigtriangleup = 1) = \pi.
\]
\subsection*{Mixture Density}
\[
f(y;\pi, \mu_1, \mu_2, \sigma^2) = (1 - \pi)f_1(y;\mu_1, \sigma^2) + \pi(y;\mu_2, \sigma^2)
\]
where \(f_1\) and \(f_2\) are the respective PDFs of the normal distributions.
\subsection*{Log-likelihood}
\[
\ell(\theta;y) = \sum_{i=1}^{n}log((1 - \pi)f_1(y_i;\mu_1, \sigma^2) + \pi f_2(y_i;\mu_2, \sigma^2))
\]
\subsection*{Latent Variable}
Let $\Delta_i$ be latent variables such that if $\Delta_i = 1$ then observation $i \sim \mathcal{N}(\mu_2, \sigma^2)$.\\
With this missing information, the likelihood becomes easier to deal with:
\[
f(y_i; \pi, \mu_1, \mu_2, \sigma^2, \Delta_i)
= \left[(1 - \pi) f_1(y; \mu_1, \sigma^2)\right]^{1 - \Delta_i}
  \left[\pi f_2(y; \mu_2, \sigma^2)\right]^{\Delta_i}.
\]
\subsection*{Latent Variable Log-likelihood}

\[
\begin{aligned}
\ell(\theta; \mathbf{y}, \Delta)
&= \sum_{i=1}^n (1 - \Delta_i) \log\left( (1 - \pi) f_1(y_i; \mu_1, \sigma^2) \right) + \Delta_i \log\left( \pi f_2(y_i; \mu_2, \sigma^2) \right) \\
&= \sum_{i=1}^n (1 - \Delta_i) \log(1 - \pi) + (1 - \Delta_i) \log\left( f_1(y_i; \mu_1, \sigma^2) \right) \\
&\quad + \sum_{i=1}^n \Delta_i \log(\pi) + \Delta_i \log\left( f_2(y_i; \mu_2, \sigma^2) \right)
\end{aligned}
\]

\subsection*{Responsibilities}
Since we don't know $\Delta_i$, we use the expected value
\[
\gamma_i(\theta) = \mathbb{E}[\Delta_i \mid \theta, \mathbf{y}] = \mathbb{P}[\Delta_i = 1 \mid \theta, \mathbf{y}].
\]

\subsection*{EM Algorithm for Two-component Normal Distribution}
\begin{enumerate}[label=\textbf{\arabic*.}]
    \item Make initial guesses of parameters: 
    $\hat{\pi}, \hat{\mu}_1, \hat{\mu}_2, \hat{\sigma}_1^2, \hat{\sigma}_2^2$.
  
    \item \textit{Expectation step:} compute the responsibilities:
    \[
    \hat{\gamma}_i = 
    \frac{\hat{\pi} f(y_i; \hat{\mu}_2, \hat{\sigma}^2)}
    {(1 - \hat{\pi}) f(y_i; \hat{\mu}_1, \hat{\sigma}^2) + \hat{\pi} f(y_i; \hat{\mu}_2, \hat{\sigma}^2)}
    \]
  
    \item \textit{Maximisation step:} estimate the parameters:
    \[
    \hat{\mu}_1 = \frac{\sum_{i=1}^n (1 - \hat{\gamma}_i) y_i}{\sum_{i=1}^n (1 - \hat{\gamma}_i)}, 
    \qquad
    \hat{\mu}_2 = \frac{\sum_{i=1}^n \hat{\gamma}_i y_i}{\sum_{i=1}^n \hat{\gamma}_i},
    \]
    \[
    \hat{\sigma}^2 = \frac{
      \sum_{i=1}^n (1 - \hat{\gamma}_i)(y_i - \hat{\mu}_1)^2 
      + \hat{\gamma}_i (y_i - \hat{\mu}_2)^2
    }{n - 2},
    \]
    \[
    \text{and} \qquad 
    \hat{\pi} = \frac{\sum_{i=1}^n \hat{\gamma}_i}{n}.
    \]
  
    \item Iterate steps 2 and 3 until convergence.
  \end{enumerate}
\end{multicols}
